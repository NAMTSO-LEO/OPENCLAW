```python
def descriptive_stats(self, variables):
    return self.data[variables].describe()

# t检验
def t_test(self, var, group):
    from scipy import stats
    g1 = self.data[self.data[group]==0][var]
    g2 = self.data[self.data[group]==1][var]
    return stats.ttest_ind(g1, g2)
```
